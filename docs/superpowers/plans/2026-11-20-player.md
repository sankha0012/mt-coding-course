# The lesson player: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the smallest player that lets the owner complete SQL lesson 1 in a local web page, with the correction ladder, narration, returns, and progress that survives a restart.

**Architecture:** A server-side state machine (`player/engine.py`) holds the sitting and knows nothing about the web or the database file. `player/store.py` keeps progress in SQLite. `player/app.py` glues them and serves four JSON routes. `player/web/` is a dumb renderer in plain HTML and JavaScript. The learner's SQL runs through `player/sql.py`, a read-only sandbox built on `checker.data`.

**Tech Stack:** Python 3.12, `uv`, FastAPI, uvicorn, `pyyaml`, stdlib `sqlite3`; `pytest` and `httpx` for tests; plain HTML, CSS and JavaScript with no build step.

**Spec:** `docs/superpowers/specs/2026-11-20-player-design.md`. Read it first. It also depends on `2026-11-20-lesson-format-and-checker-design.md`.

## Global Constraints

- Python 3.12 and `uv`. New runtime packages: `fastapi`, `uvicorn`. New dev package: `httpx`. Nothing else.
- Lesson text never lives in `player/`. The course-wide teacher lines live in `content/ui.yaml`. Button labels and other chrome live in `player/web/`.
- The server listens on `127.0.0.1` only, default port 8770.
- No timer, score, streak, percentage, step counter, progress bar, notes, reference panel, glossary, multiple choice, red, or the word "wrong" anywhere on screen or in `content/ui.yaml`.
- The progress file is never deleted, overwritten, or reset by any code. No `DELETE` or `DROP` statement exists in `player/`. Tests use a temporary folder, never `progress/`.
- Progress rows are only appended, except `runs.state_json` and `runs.finished_at`, which are updated.
- The learner's SQL is `SELECT` only, stopped after 5 seconds, and at most 20 rows are sent to the page.
- Commit messages are normal English prose and end with the attribution line the session gives for commits.
- Work on branch `milestone-4-player`, stacked on `milestone-4-returns`. Do not push, merge, or open a PR without the owner's yes.
- This plan, once approved, is the owner's permission to write the files named in it and no others.

Create `player/sql.py`:

```python
import re
import sqlite3
import threading
import time
from dataclasses import dataclass, field

from checker.data import load_db

_ALLOWED = {sqlite3.SQLITE_SELECT, sqlite3.SQLITE_READ, sqlite3.SQLITE_FUNCTION, sqlite3.SQLITE_RECURSIVE}
_SINGLE_QUOTED = re.compile(r"('(?:[^']|'')*')")
_DOUBLE_QUOTED = re.compile(r'"([^"]*)"')
_TOKEN = re.compile(r"'(?:[^']|'')*'|\w+|[^\s\w]")
_lock = threading.Lock()


def _authorizer(action, *_):
    return sqlite3.SQLITE_OK if action in _ALLOWED else sqlite3.SQLITE_DENY


def fix_quotes(text):
    """A word in double quotes is text in this course. Single-quoted text is left alone."""
    parts = _SINGLE_QUOTED.split(text)
    return "".join(
        p if i % 2 else _DOUBLE_QUOTED.sub(lambda m: "'" + m.group(1).replace("'", "''") + "'", p)
        for i, p in enumerate(parts))


def normalise(text):
    """Text in a form that is equal when only case, spacing, quote style or a final semicolon differ."""
    t = re.sub(r";\s*$", "", fix_quotes(text).strip())
    return " ".join(tok if tok.startswith("'") else tok.lower() for tok in _TOKEN.findall(t))


@dataclass
class Result:
    error: str | None
    columns: list
    rows: list
    total: int
    full: list = field(default_factory=list, repr=False)

    def item(self, label=None):
        out = {"type": "result", "columns": self.columns, "rows": self.rows,
               "total": self.total, "error": self.error}
        if label:
            out["label"] = label
        return out


def open_database(data_dir):
    conn = load_db(data_dir, check_same_thread=False)
    conn.set_authorizer(_authorizer)
    return conn


def run_sql(conn, text, limit=20, timeout=5.0):
    sql = fix_quotes(text)
    deadline = time.monotonic() + timeout
    with _lock:
        conn.set_progress_handler(lambda: 1 if time.monotonic() > deadline else 0, 10_000)
        try:
            cur = conn.execute(sql)
            full = cur.fetchall()
            columns = [d[0] for d in cur.description] if cur.description else []
        except sqlite3.OperationalError as e:
            msg = "That took too long, so it was stopped." if "interrupted" in str(e) else str(e)
            return Result(msg, [], [], 0)
        except (sqlite3.Error, sqlite3.Warning) as e:
            return Result(str(e), [], [], 0)
        finally:
            conn.set_progress_handler(None, 0)
    return Result(None, columns, [list(r) for r in full[:limit]], len(full), full)
```

```python
import copy

from checker.data import same_rows

from .sql import normalise, run_sql


def _teacher(text, style=None):
    item = {"type": "teacher", "text": text}
    if style:
        item["style"] = style
    return item


class Engine:
    """The sitting as a state machine. It knows neither the web nor the progress file."""

    def __init__(self, steps, conn, ui, rng):
        self.steps, self.conn, self.ui, self.rng = steps, conn, ui, rng
        self._cache = {}

    # ---- public

    def start(self):
        st = {"pos": 0, "phase": "continue", "playing": {"step": 0, "variant": "main"},
              "misses": 0, "rung": 0, "why": None, "queue": [], "returned": [],
              "last_narration": None, "last_ui": {}}
        items = []
        self._enter(st, items)
        return st, items, []

    def pending(self, st):
        phase, step = st["phase"], st["playing"]["step"]
        if phase == "run":
            return {"mode": "run", "sql": self.steps[step]["shows"][-1]}
        if phase == "answer":
            return {"mode": "answer", "step": step + 1, "variant": st["playing"]["variant"]}
        return {"mode": phase}

    def advance(self, state):
        st, items = copy.deepcopy(state), []
        cur = self.steps[st["playing"]["step"]]
        if st["phase"] == "run":
            sql = cur["shows"][-1]
            items += [{"type": "learner", "sql": sql}, run_sql(self.conn, sql).item()]
            if cur.get("after_run"):
                items.append(_teacher(cur["after_run"].strip()))
            st["phase"] = "continue"
        elif st["phase"] == "continue":
            if st["playing"]["variant"] == "main":
                st["pos"] = st["playing"]["step"] + 1
            self._enter(st, items)
        else:
            raise ValueError("nothing to continue")
        return st, items, []

    def answer(self, state, text):
        st = copy.deepcopy(state)
        if st["phase"] != "answer":
            raise ValueError("not waiting for an answer")
        cur, step, variant = self._cur(st), st["playing"]["step"], st["playing"]["variant"]
        res = run_sql(self.conn, text)
        items = [{"type": "learner", "sql": text}, res.item()]
        before = st["misses"]

        def event(outcome):
            return {"step": step + 1, "variant": variant, "typed": text, "outcome": outcome, "misses": before}

        if self._correct(cur, res):
            outcome = self._on_correct(st, cur, step, variant, text, items)
            return st, items, [event(outcome)]
        return st, items, [event("miss")]

    # ---- moving through the lesson

    def _enter(self, st, items):
        j = st["pos"]
        if j >= len(self.steps):
            st["phase"] = "done"
            return
        self._begin_step(st, items, j)

    def _begin_step(self, st, items, j):
        s = self.steps[j]
        st.update(playing={"step": j, "variant": "main"}, misses=0, rung=0, why=None)
        style = s["kind"] if s.get("kind") in ("aside", "recap") else None
        items.append(_teacher(s["text"].strip(), style))
        if "model" in s:
            st["phase"] = "answer"
        elif s.get("shows"):
            st["phase"] = "run"
        else:
            st["phase"] = "continue"

    def _cur(self, st):
        s = self.steps[st["playing"]["step"]]
        return s["return"] if st["playing"]["variant"] == "return" else s

    # ---- judging an answer

    def _rows(self, sql):
        if sql not in self._cache:
            r = run_sql(self.conn, sql)
            self._cache[sql] = None if r.error else r.full
        return self._cache[sql]

    @staticmethod
    def _flags(cur):
        return {"ordered": cur.get("ordered", False), "any_column_order": cur.get("any_column_order", False)}

    def _correct(self, cur, res):
        if res.error:
            return False
        for sql in [cur["model"], *cur.get("accepts", [])]:
            want = self._rows(sql)
            if want is not None and same_rows(res.full, want, **self._flags(cur)):
                return True
        return False

    def _on_correct(self, st, cur, step, variant, text, items):
        if st["why"]:
            items.append(_teacher(st["why"][0].upper() + st["why"][1:]))
        if normalise(text) == normalise(cur["model"]) and text.strip() != cur["model"].strip():
            items.append({"type": "compare", "yours": text.strip(), "usual": cur["model"]})
            items.append(_teacher(self._pick(st, "slip")))
            outcome = "slip"
        else:
            self._narrate(st, cur, items)
            outcome = "correct"
        st["phase"] = "continue"
        return outcome

    def _narrate(self, st, cur, items):
        pool = list(cur.get("narration", []))
        if not pool:
            return
        line = self.rng.choice([x for x in pool if x != st["last_narration"]] or pool)
        st["last_narration"] = line
        if line.strip():
            items.append(_teacher(line))

    def _pick(self, st, key):
        options = self.ui[key]
        line = self.rng.choice([x for x in options if x != st["last_ui"].get(key)] or options)
        st["last_ui"][key] = line
        return line
```

```python
    # ---- the correction ladder

    def _match_wrong(self, cur, text, res):
        mine = normalise(text)
        for w in cur.get("wrong", []):
            if normalise(w["sql"]) == mine:
                return w
            if res.error is None:
                want = self._rows(w["sql"])
                if want is not None and same_rows(res.full, want, **self._flags(cur)):
                    return w
        return None

    def _next_rung(self, st, cur, text, res):
        r = st["rung"]
        if r == 0:
            return 1
        if r == 1 and self._match_wrong(cur, text, res):
            return 2
        if r <= 2 and cur.get("walk"):
            return 3
        return 4

    def _on_miss(self, st, cur, text, res, items):
        st["misses"] += 1
        rung = st["rung"] = self._next_rung(st, cur, text, res)
        if rung == 1:
            items.append(_teacher(self._pick(st, "again")))
            if cur.get("thinking_order"):
                items.append(_teacher(cur["thinking_order"]))
        elif rung == 2:
            w = self._match_wrong(cur, text, res)
            items.append(_teacher(w.get("ask") or w["tell"]))
            st["why"] = w.get("why")
        elif rung == 3:
            steps = "\n".join(f"{i}. {w}" for i, w in enumerate(cur["walk"], 1))
            items.append(_teacher(f"{self._pick(st, 'walk_intro')}\n\n{steps}"))
        else:
            items.append(_teacher(f"{self._pick(st, 'show_answer')}\n\n```sql\n{cur['model']}\n```"))
```

```python
import json
import sqlite3
from datetime import datetime, timezone
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS runs (
  id INTEGER PRIMARY KEY, lesson TEXT, started_at TEXT, finished_at TEXT, state_json TEXT);
CREATE TABLE IF NOT EXISTS items (
  run_id INTEGER, seq INTEGER, json TEXT, PRIMARY KEY (run_id, seq));
CREATE TABLE IF NOT EXISTS events (
  id INTEGER PRIMARY KEY, run_id INTEGER, ts TEXT, step INTEGER, variant TEXT,
  typed TEXT, outcome TEXT, misses INTEGER);
"""


def _now():
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


class Store:
    """The owner's progress. Rows are only added. State and finish time on a run are updated."""

    def __init__(self, folder):
        Path(folder).mkdir(parents=True, exist_ok=True)
        # ponytail: one shared connection, guarded by the Player's lock. Fine for one local learner.
        self.conn = sqlite3.connect(Path(folder) / "progress.db", check_same_thread=False)
        self.conn.executescript(SCHEMA)

    def latest(self, lesson):
        row = self.conn.execute(
            "SELECT id, state_json FROM runs WHERE lesson = ? ORDER BY id DESC LIMIT 1", (lesson,)).fetchone()
        return (row[0], json.loads(row[1])) if row else None

    def new_run(self, lesson, state, items, finished=False):
        with self.conn:
            cur = self.conn.execute(
                "INSERT INTO runs (lesson, started_at, finished_at, state_json) VALUES (?, ?, ?, ?)",
                (lesson, _now(), _now() if finished else None, json.dumps(state)))
            self._add_items(cur.lastrowid, items)
        return cur.lastrowid

    def apply(self, run_id, state, items, events, finished=False):
        with self.conn:
            self._add_items(run_id, items)
            self.conn.executemany(
                "INSERT INTO events (run_id, ts, step, variant, typed, outcome, misses) VALUES (?, ?, ?, ?, ?, ?, ?)",
                [(run_id, _now(), e["step"], e["variant"], e["typed"], e["outcome"], e["misses"]) for e in events])
            self.conn.execute(
                "UPDATE runs SET state_json = ?, finished_at = COALESCE(finished_at, ?) WHERE id = ?",
                (json.dumps(state), _now() if finished else None, run_id))

    def _add_items(self, run_id, items):
        last = self.conn.execute(
            "SELECT COALESCE(MAX(seq), -1) FROM items WHERE run_id = ?", (run_id,)).fetchone()[0]
        self.conn.executemany(
            "INSERT INTO items (run_id, seq, json) VALUES (?, ?, ?)",
            [(run_id, last + 1 + i, json.dumps(item, ensure_ascii=False)) for i, item in enumerate(items)])

    def items(self, run_id):
        rows = self.conn.execute("SELECT json FROM items WHERE run_id = ? ORDER BY seq", (run_id,))
        return [json.loads(r[0]) for r in rows]

    def events(self):
        cur = self.conn.execute(
            "SELECT id, run_id, ts, step, variant, typed, outcome, misses FROM events ORDER BY id")
        names = [d[0] for d in cur.description]
        return [dict(zip(names, r)) for r in cur.fetchall()]
```

```python
import random
import threading
from pathlib import Path

import yaml
from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse, JSONResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel

from checker.lesson import load_strand

from .engine import Engine
from .sql import open_database
from .store import Store

WEB = Path(__file__).parent / "web"


class Conflict(Exception):
    pass


class Player:
    """Joins the engine, the database and the progress file."""

    def __init__(self, content_dir, data_dir, progress_dir, rng=None):
        content = Path(content_dir)
        self.thoughts, lessons = load_strand(content)
        self.lesson, self.steps = lessons[0]
        ui = yaml.safe_load((content.parent / "ui.yaml").read_text(encoding="utf-8"))
        self.engine = Engine(self.steps, open_database(data_dir), ui, rng or random.Random())
        self.store = Store(progress_dir)
        self.lock = threading.Lock()

    def _current(self):
        return self.store.latest(self.lesson) or self._start()

    def _start(self):
        st, items, _ = self.engine.start()
        return self.store.new_run(self.lesson, st, items, finished=st["phase"] == "done"), st

    def _view(self, run_id, st):
        return {"run_id": run_id, "items": self.store.items(run_id), "pending": self.engine.pending(st)}

    def view(self):
        with self.lock:
            return self._view(*self._current())

    def answer(self, text):
        with self.lock:
            run_id, st = self._current()
            if st["phase"] != "answer":
                raise Conflict("not waiting for an answer")
            st, items, events = self.engine.answer(st, text)
            self.store.apply(run_id, st, items, events, finished=st["phase"] == "done")
            return self._view(run_id, st)

    def next(self):
        with self.lock:
            run_id, st = self._current()
            if st["phase"] not in ("run", "continue"):
                raise Conflict("nothing to continue")
            st, items, events = self.engine.advance(st)
            self.store.apply(run_id, st, items, events, finished=st["phase"] == "done")
            return self._view(run_id, st)

    def new_run(self):
        with self.lock:
            if self._current()[1]["phase"] != "done":
                raise Conflict("the sitting is not over")
            return self._view(*self._start())


class Answer(BaseModel):
    sql: str


def make_app(content_dir, data_dir, progress_dir, rng=None):
    player = Player(content_dir, data_dir, progress_dir, rng)
    app = FastAPI()

    @app.exception_handler(Conflict)
    def conflict(_, exc):
        return JSONResponse({"detail": str(exc)}, status_code=409)

    @app.get("/api/run")
    def get_run():
        return player.view()

    @app.post("/api/answer")
    def post_answer(a: Answer):
        if not a.sql.strip():
            raise HTTPException(400, "empty answer")
        return player.answer(a.sql)

    @app.post("/api/next")
    def post_next():
        return player.next()

    @app.post("/api/run/new")
    def post_new():
        return player.new_run()

    @app.get("/")
    def index():
        return FileResponse(WEB / "index.html")

    app.mount("/static", StaticFiles(directory=WEB), name="static")
    return app
```

```javascript
const itemsEl = document.getElementById('items');
const inputEl = document.getElementById('input');

async function api(path, body) {
  const options = path === '/api/run' ? {} : {
    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body || {}) };
  const r = await fetch(path, options);
  if (!r.ok) throw new Error(path + ' ' + r.status);
  return r.json();
}

function el(tag, cls, text) {
  const e = document.createElement(tag);
  if (cls) e.className = cls;
  if (text !== undefined) e.textContent = text;
  return e;
}

function lineNodes(text) {
  const out = document.createDocumentFragment();
  text.split('\n').forEach((line, i) => {
    if (i) out.appendChild(document.createElement('br'));
    line.split('`').forEach((part, j) => out.appendChild(j % 2 ? el('code', null, part) : document.createTextNode(part)));
  });
  return out;
}

function renderText(text, cls) {
  const box = el('div', 'teacher' + (cls ? ' ' + cls : ''));
  text.split(/```sql\n([\s\S]*?)\n```/).forEach((part, i) => {
    if (i % 2) { box.appendChild(el('pre', null, part)); return; }
    part.split(/\n\s*\n/).filter(p => p.trim()).forEach(p => {
      const para = el('p'); para.appendChild(lineNodes(p.trim())); box.appendChild(para);
    });
  });
  return box;
}

function renderResult(item) {
  const box = el('div');
  if (item.label) box.appendChild(el('div', 'label', item.label));
  if (item.error) { box.appendChild(el('pre', 'output', item.error)); return box; }
  const table = el('table');
  const head = el('tr');
  item.columns.forEach(c => head.appendChild(el('th', null, c)));
  table.appendChild(head);
  item.rows.forEach(r => {
    const tr = el('tr');
    r.forEach(v => tr.appendChild(el('td', null, v === null ? '' : String(v))));
    table.appendChild(tr);
  });
  box.appendChild(table);
  if (item.total > item.rows.length) box.appendChild(el('div', 'total', item.total.toLocaleString() + ' rows in all'));
  return box;
}

function renderItem(item) {
  if (item.type === 'teacher') return renderText(item.text, item.style);
  if (item.type === 'learner') return el('pre', 'learner', item.sql);
  if (item.type === 'result') return renderResult(item);
  const box = el('div', 'compare');
  [['Yours', item.yours], ['Usual', item.usual]].forEach(([name, sql]) => {
    const col = el('div'); col.appendChild(el('div', 'caption', name)); col.appendChild(el('pre', null, sql)); box.appendChild(col);
  });
  return box;
}

function button(label, onClick) {
  const b = el('button', null, label); b.type = 'button'; b.onclick = onClick; return b;
}

function renderInput(pending) {
  inputEl.replaceChildren();
  if (pending.mode === 'continue') {
    inputEl.appendChild(button('Continue', () => send('/api/next')));
  } else if (pending.mode === 'run') {
    inputEl.appendChild(el('pre', null, pending.sql));
    inputEl.appendChild(button('Run', () => send('/api/next')));
  } else if (pending.mode === 'answer') {
    const box = el('textarea');
    box.setAttribute('autocomplete', 'off'); box.setAttribute('autocapitalize', 'off');
    box.setAttribute('autocorrect', 'off'); box.setAttribute('spellcheck', 'false');
    const hint = el('p', 'hint');
    const run = () => {
      if (!box.value.trim()) { hint.textContent = 'Type an answer first.'; return; }
      send('/api/answer', { sql: box.value });
    };
    box.oninput = () => { hint.textContent = ''; };
    box.onkeydown = e => { if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') { e.preventDefault(); run(); } };
    inputEl.append(box, button('Run', run), hint);
    box.focus();
  } else if (pending.mode === 'done') {
    inputEl.appendChild(button('Sit this lesson again', () => send('/api/run/new')));
  }
}

function render(view) {
  itemsEl.replaceChildren();
  const lastTeacher = view.items.map(i => i.type).lastIndexOf('teacher');
  view.items.forEach((item, i) => {
    const node = renderItem(item);
    if (i < lastTeacher) node.classList.add('old');
    itemsEl.appendChild(node);
  });
  renderInput(view.pending);
  inputEl.scrollIntoView({ block: 'end', behavior: 'smooth' });
}

async function send(path, body) {
  try { render(await api(path, body)); }
  catch (e) { inputEl.appendChild(el('p', 'hint', 'That did not go through. Reload the page.')); }
}

api('/api/run').then(render);
```

```python
import argparse
from collections import Counter

from checker.lesson import load_strand

from .store import Store


def build_report(events, steps):
    """One line per step and variant, with the most misses first."""
    groups = {}
    for e in events:
        groups.setdefault((e["step"], e["variant"]), []).append(e)
    if not groups:
        return ["No answers yet."]
    rows = []
    for (step, variant), evs in groups.items():
        misses = [e for e in evs if e["outcome"] == "miss"]
        first_try = sum(1 for e in evs if e["outcome"] in ("correct", "slip") and e["misses"] == 0)
        common = Counter(e["typed"] for e in misses).most_common(1)
        needs = ", ".join(steps[step - 1].get("requires", [])) if step <= len(steps) else ""
        name = f"step {step}" + (" (return)" if variant == "return" else "")
        line = f"{name}: {len(evs)} answers, {first_try} right the first time, {len(misses)} misses"
        if common:
            line += f"; most common miss: {common[0][0]}"
        if needs:
            line += f"; needs: {needs}"
        rows.append((-len(misses), step, variant, line))
    return [r[3] for r in sorted(rows)]


def main(argv=None):
    p = argparse.ArgumentParser(prog="player.report", description="Where the learner missed.")
    p.add_argument("--content", default="content/sql")
    p.add_argument("--progress", default="progress")
    args = p.parse_args(argv)
    _, lessons = load_strand(args.content)
    print("\n".join(build_report(Store(args.progress).events(), lessons[0][1])))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
from pathlib import Path

import pytest
from fastapi.testclient import TestClient

from checker.lesson import load_strand
from player.app import make_app
from player.store import Store

ROOT = Path(__file__).parent.parent
CONTENT = ROOT / "content" / "sql"


@pytest.fixture
def lesson():
    return load_strand(CONTENT)[1][0][1]


def play(client, lesson, miss_first):
    """Play the whole lesson through the API. Returns the final view."""
    body = client.get("/api/run").json()
    for _ in range(2000):
        pending = body["pending"]
        if pending["mode"] == "done":
            return body
        if pending["mode"] in ("run", "continue"):
            body = client.post("/api/next").json()
            continue
        step = lesson[pending["step"] - 1]
        model = (step["return"] if pending["variant"] == "return" else step)["model"]
        if miss_first and step.get("kind") != "cold-check":
            body = client.post("/api/answer", json={"sql": "SELECT 1"}).json()
            assert body["pending"]["mode"] == "answer"
        body = client.post("/api/answer", json={"sql": model}).json()
    raise AssertionError("the lesson never finished")


def test_all_of_lesson_1_can_be_played_with_the_model_answers(data_dir, tmp_path, lesson):
    client = TestClient(make_app(CONTENT, data_dir, tmp_path))
    body = play(client, lesson, miss_first=False)
    assert body["pending"] == {"mode": "done"}
    events = Store(tmp_path).events()
    assert events and all(e["outcome"] in ("correct", "shown") for e in events)
    assert all(e["variant"] == "main" for e in events)


def test_lesson_1_can_be_played_missing_every_step_once(data_dir, tmp_path, lesson):
    client = TestClient(make_app(CONTENT, data_dir, tmp_path))
    body = play(client, lesson, miss_first=True)
    assert body["pending"] == {"mode": "done"}
    events = Store(tmp_path).events()
    returns = [e for e in events if e["variant"] == "return" and e["outcome"] == "correct"]
    with_return = [s for s in lesson if "return" in s]
    assert len(returns) == len(with_return) == 15
    assert not any("wrong" in str(i).lower() for i in body["items"] if i["type"] == "teacher")


def test_sitting_again_starts_a_new_run_and_keeps_the_old(data_dir, tmp_path, lesson):
    client = TestClient(make_app(CONTENT, data_dir, tmp_path))
    first = play(client, lesson, miss_first=False)
    second = client.post("/api/run/new").json()
    assert second["run_id"] > first["run_id"]
    assert second["pending"]["mode"] == "run"
    assert Store(tmp_path).items(first["run_id"]) == first["items"]
```
